Image Compression Function

library(readr)

compress_image <- function(img_matrix, k) {
  # 1. Calculate the cross product V = X'X
  V <- t(img_matrix) %*% img_matrix
  
  # 2. Get the eigendecomposition of V 
  # U becomes our matrix of eigenvectors
  decomp <- eigen(V)
  U <- decomp$vectors
  
  # 3. Calculate the principal component scores (Z = XU)
  Z <- img_matrix %*% U
  
  # 4. Create the rank k approximation
  # We subset the first k columns of Z and U
  Z_k <- Z[, 1:k, drop = FALSE]
  U_k <- U[, 1:k, drop = FALSE]
  
  # X^(k) = Z_k * U_k'
  X_k <- Z_k %*% t(U_k)
  
  # 5. Calculate the Frobenius norm error
  approx_error <- norm(img_matrix - X_k, type = "F")
  
  # 6. Return both pieces of data as a list
  return(list(
    approximation = X_k,
    error = approx_error
  ))
}

# --- HELPER FUNCTION TO DRAW IMAGES ---
# R draws matrices sideways by default; this transposes and flips it upright
draw_image <- function(mat, plot_title) {
  image(t(mat)[, nrow(mat):1], col = gray.colors(256), main = plot_title, axes = FALSE)
}

# ==========================================
# REQUIREMENT 1: PLOT 10 APPROXIMATIONS
# ==========================================

# Set up a grid layout to show 12 smaller images at once (3 rows, 4 columns)
par(mfrow = c(3, 4)) 
par(mar = c(1, 1, 2, 1)) # Shrink the margins so they fit nicely

# Draw the original image first
draw_image(x, "Original Image")

# Loop through k = 1 to 10 and draw each compressed version
for (k_val in 1:10) {
  result <- compress_image(x, k_val)
  draw_image(result$approximation, paste("k =", k_val))
}


# ==========================================
# REQUIREMENT 2: PLOT ERROR VS. K
# ==========================================

# Find the maximum possible k (which is the total number of columns in the image)
max_k <- ncol(x)
errors <- numeric(max_k) # Create an empty vector to store the errors

# Loop through every possible k and save the error
for (k_val in 1:max_k) {
  errors[k_val] <- compress_image(x, k_val)$error
}

# Reset the plot layout to a single, normal-sized graph
par(mfrow = c(1, 1)) 

par(mar = c(5, 4, 4, 2) + 0.1) # Reset to default margins

# Draw the line plot
plot(1:max_k, errors, type = "b", col = "blue", pch = 16,
     xlab = "k (Rank of Approximation)", 
     ylab = "Frobenius Error", 
     main = "Image 1: Approximation Error vs. k")
